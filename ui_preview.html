1. AnswerOptionButton (Карточка варианта ответа) 
@Composable
fun AnswerOptionCard(
    letter: String,
    title: String,
    artist: String,
    state: AnswerState, // Default, Correct, Wrong
    onClick: () -> Unit
) {
    val borderColor = when (state) {
        AnswerState.Correct -> AnswerCorrect
        AnswerState.Wrong -> AnswerWrong
        else -> MaterialTheme.colorScheme.outline
    }
    val containerColor = when (state) {
        AnswerState.Correct -> AnswerCorrect.copy(alpha = 0.15f)
        AnswerState.Wrong -> AnswerWrong.copy(alpha = 0.15f)
        else -> MaterialTheme.colorScheme.surface
    }

    Surface(
        onClick = onClick,
        shape = RoundedCornerShape(16.dp),
        color = containerColor,
        border = BorderStroke(1.5.dp, borderColor),
        modifier = Modifier.fillMaxWidth()
    ) {
        Row(
            modifier = Modifier.padding(14.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            // Буква A, B, C, D
            Box(
                modifier = Modifier
                    .size(32.dp)
                    .background(MaterialTheme.colorScheme.surfaceVariant, RoundedCornerShape(8.dp)),
                contentAlignment = Alignment.Center
            ) {
                Text(text = letter, fontWeight = FontWeight.Bold, color = TextPrimary)
            }
            Spacer(modifier = Modifier.width(12.dp))
            Column {
                Text(text = title, style = MaterialTheme.typography.titleMedium)
                Text(text = artist, style = MaterialTheme.typography.bodySmall, color = TextSecondary)
            }
        }
    }
}
💡 Советы по UX/UI анимациям в Android
UX Guidelines
  • Плавный таймер: используйте animateFloatAsState для полоски времени, чтобы она не дергалась каждую секунду.
  • Визуальный отклик: при нажатии на карточку варианта подсвечивайте правильный ответ зеленым (AnswerCorrect), а ошибочный — красным (AnswerWrong) с задержкой 1.2 сек перед переходом к следующему треку.
  • Edge-to-Edge: в MainActivity.kt у вас уже включен enableEdgeToEdge() — используйте Modifier.safeDrawingPadding() для корректных отступов под вырез и панель навигации.
  • Пластинка и эквалайзер: анимируйте вращение обложки через rememberInfiniteTransition() при воспроизведении отрывка из AudioPlayer.
